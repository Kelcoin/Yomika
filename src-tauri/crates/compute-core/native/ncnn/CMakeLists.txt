cmake_minimum_required(VERSION 3.22)
project(yomika_ncnn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ncnn_DIR "${NCNN_SDK_DIR}/${ANDROID_ABI}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
add_library(yomika_ncnn SHARED shim.cpp litert.cpp)
target_link_libraries(yomika_ncnn PRIVATE ncnn log dl)
target_include_directories(yomika_ncnn PRIVATE ../litert/include)
target_link_options(yomika_ncnn PRIVATE "-Wl,-z,max-page-size=16384")
if(YOMIKA_NCNN_SMOKE)
  add_executable(npu_smoke npu_smoke.cpp)
  target_link_libraries(npu_smoke PRIVATE yomika_ncnn)
  add_executable(nnapi_lifetime_smoke nnapi_lifetime_smoke.cpp)
  target_link_libraries(nnapi_lifetime_smoke PRIVATE dl)
  add_executable(ncnn_smoke smoke.cpp)
  target_link_libraries(ncnn_smoke PRIVATE yomika_ncnn)
endif()
target_compile_options(yomika_ncnn PRIVATE -fexceptions)
get_target_property(ncnn_options ncnn INTERFACE_COMPILE_OPTIONS)
list(REMOVE_ITEM ncnn_options "-fno-exceptions")
set_property(TARGET ncnn PROPERTY INTERFACE_COMPILE_OPTIONS "${ncnn_options}")
